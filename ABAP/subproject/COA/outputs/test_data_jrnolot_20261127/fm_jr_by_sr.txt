LZBARRIERU01
FUNCTION ZQM_GET_BATCH_JR_BY_SR.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(CHARG_S) TYPE  CHARG_D
*"  EXPORTING
*"     REFERENCE(MATNR_J) TYPE  MATNR
*"     REFERENCE(CHARG_J) TYPE  CHARG_D
*"----------------------------------------------------------------------
  DATA: BEGIN OF IT_BATCH OCCURS 0,
           MBLNR LIKE AUFM-MBLNR,
           MJAHR LIKE AUFM-MJAHR,
           MATNR LIKE AUFM-MATNR,
           WERKS LIKE AUFM-WERKS,
           CHARG LIKE AUFM-CHARG,
           BWART LIKE AUFM-BWART,
           AUFNR LIKE AUFM-AUFNR,
           MENGE LIKE AUFM-MENGE,
           MEINS LIKE AUFM-MEINS,
           SMBLN LIKE MSEG-SMBLN,
           SJAHR LIKE MSEG-SJAHR,
           SMBLP LIKE MSEG-SMBLP,
         END OF IT_BATCH.
  DATA: IT_BATCH_JR LIKE IT_BATCH OCCURS 0 WITH HEADER LINE.
  DATA: IT_BATCH_JC LIKE IT_BATCH OCCURS 0 WITH HEADER LINE.
  DATA: IT_BATCH_SR LIKE IT_BATCH OCCURS 0 WITH HEADER LINE.
  DATA: IT_BATCH_SC LIKE IT_BATCH OCCURS 0 WITH HEADER LINE.
  DATA: IT_BATCH_CG LIKE IT_BATCH OCCURS 0 WITH HEADER LINE.
  DATA: IT_BATCH_CC LIKE IT_BATCH OCCURS 0 WITH HEADER LINE.
  DATA: IT_AFPO LIKE AFPO OCCURS 0 WITH HEADER LINE.
  DATA: V_ABLAD TYPE ABLAD.
  REFRESH: IT_BATCH_JR, IT_BATCH_JC, IT_BATCH_SR, IT_BATCH_SC, IT_AFPO.
  CLEAR: IT_BATCH_JR, IT_BATCH_JC, IT_BATCH_SR, IT_BATCH_SC, IT_AFPO, V_ABLAD.
  SELECT
    AUFM~MBLNR
    AUFM~MJAHR
    AUFM~MATNR
    AUFM~WERKS
    AUFM~CHARG
    AUFM~BWART
    AUFM~AUFNR
    AUFM~MENGE
    AUFM~MEINS
    MSEG~SMBLN
    MSEG~SJAHR
    MSEG~SMBLP
  APPENDING CORRESPONDING FIELDS OF TABLE IT_BATCH_SR
  FROM AUFM
  JOIN MSEG ON AUFM~MANDT  =  MSEG~MANDT  AND
               AUFM~MBLNR  =  MSEG~MBLNR  AND
               AUFM~MJAHR  =  MSEG~MJAHR  AND
               AUFM~ZEILE  =  MSEG~ZEILE
  WHERE AUFM~CHARG EQ CHARG_S
    AND AUFM~BWART EQ '101' %_HINTS ORACLE 'INDEX("MESG" "MSEG~Z07")'.
  IF IT_BATCH_SR[] IS INITIAL.
    EXIT.
  ENDIF.
  "Get data batch cancelation
  SELECT
    AUFM~MBLNR
    AUFM~MJAHR
    AUFM~MATNR
    AUFM~WERKS
    AUFM~CHARG
    AUFM~BWART
    AUFM~AUFNR
    AUFM~MENGE
    AUFM~MEINS
    MSEG~SMBLN
    MSEG~SJAHR
    MSEG~SMBLP
  APPENDING CORRESPONDING FIELDS OF TABLE IT_BATCH_SC
  FROM AUFM
  JOIN MSEG ON AUFM~MANDT  =  MSEG~MANDT  AND
               AUFM~MBLNR  =  MSEG~MBLNR  AND
               AUFM~MJAHR  =  MSEG~MJAHR
  FOR ALL ENTRIES IN IT_BATCH_SR
  WHERE MSEG~SMBLN = IT_BATCH_SR-MBLNR
    AND MSEG~SJAHR = IT_BATCH_SR-MJAHR %_HINTS ORACLE 'INDEX("MESG" "MSEG~Z10")'.
  " Eliminate batch cancelation
  LOOP AT IT_BATCH_SC.
    LOOP AT IT_BATCH_SR WHERE MBLNR = IT_BATCH_SC-SMBLN AND MJAHR = IT_BATCH_SC-SJAH
      DELETE IT_BATCH_SR.
    ENDLOOP.
  ENDLOOP.
  SORT IT_BATCH_SR BY CHARG.
  DELETE ADJACENT DUPLICATES FROM IT_BATCH_SR COMPARING CHARG.
  IF IT_BATCH_SR[] IS INITIAL.
    EXIT.
  ENDIF.
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE IT_AFPO
    FROM AFPO
     FOR ALL ENTRIES IN IT_BATCH_SR
   WHERE MILL_OC_AUFNR_U = IT_BATCH_SR-AUFNR.
  IF IT_AFPO[] IS INITIAL.
    "Case Change Grade
    READ TABLE IT_BATCH_SR INDEX 1.
    IF SY-SUBRC EQ 0 AND IT_BATCH_SR-AUFNR+0(1) EQ '1'.
      REFRESH: IT_BATCH_CG, IT_BATCH_CC.
      CLEAR: IT_BATCH_CG, IT_BATCH_CC.
      SELECT
        MBLNR
        MJAHR
        MATNR
        WERKS
        CHARG
        BWART
        AUFNR
        MENGE
        MEINS
        SMBLN
        SJAHR
        SMBLP
        APPENDING CORRESPONDING FIELDS OF TABLE IT_BATCH_CG
        FROM MSEG
       WHERE AUFNR EQ IT_BATCH_SR-AUFNR
         AND BWART EQ 261.
      IF IT_BATCH_CG[] IS INITIAL.
        EXIT.
      ENDIF.
      "Get data batch cancelation
      SELECT
        MBLNR
        MJAHR
        MATNR
        WERKS
        CHARG
        BWART
        AUFNR
        MENGE
        MEINS
        SMBLN
        SJAHR
        SMBLP
        APPENDING CORRESPONDING FIELDS OF TABLE IT_BATCH_CC
        FROM MSEG
        FOR ALL ENTRIES IN IT_BATCH_CG
        WHERE MSEG~SMBLN = IT_BATCH_CG-MBLNR
          AND MSEG~SJAHR = IT_BATCH_CG-MJAHR %_HINTS ORACLE 'INDEX("MESG" "MSEG~Z10"
      "Eliminate batch cancelation
      LOOP AT IT_BATCH_CC.
        LOOP AT IT_BATCH_CG WHERE MBLNR = IT_BATCH_CC-SMBLN AND MJAHR = IT_BATCH_CC-
          DELETE IT_BATCH_CG.
        ENDLOOP.
      ENDLOOP.
      READ TABLE IT_BATCH_CG INDEX 1.
      IF SY-SUBRC EQ 0.
        CALL FUNCTION 'ZQM_GET_BATCH_JR_BY_SR'
          EXPORTING
            CHARG_S = IT_BATCH_CG-CHARG
          IMPORTING
            MATNR_J = MATNR_J
            CHARG_J = CHARG_J.
      ENDIF.
    ELSE.
      EXIT.
    ENDIF.
  ENDIF.
  LOOP AT IT_AFPO.
    CLEAR: IT_BATCH_JR, IT_BATCH_JC.
    REFRESH: IT_BATCH_JR, IT_BATCH_JC.
    SELECT
      AUFM~MBLNR
      AUFM~MJAHR
      AUFM~MATNR
      AUFM~WERKS
      AUFM~CHARG
      AUFM~BWART
      AUFM~AUFNR
      AUFM~MENGE
      AUFM~MEINS
      MSEG~SMBLN
      MSEG~SJAHR
      MSEG~SMBLP
      APPENDING CORRESPONDING FIELDS OF TABLE IT_BATCH_JR
      FROM AUFM
      JOIN MSEG ON AUFM~MANDT  =  MSEG~MANDT  AND
                   AUFM~MBLNR  =  MSEG~MBLNR  AND
                   AUFM~MJAHR  =  MSEG~MJAHR  AND
                   AUFM~ZEILE  =  MSEG~ZEILE
      WHERE AUFM~AUFNR = IT_AFPO-AUFNR
        AND AUFM~BWART EQ '261' %_HINTS ORACLE 'INDEX("MESG" "MSEG~Z08")'.
    IF IT_BATCH_JR[] IS INITIAL.
      EXIT.
    ENDIF.
    " Get data batch cancelation
    SELECT
      AUFM~MBLNR
      AUFM~MJAHR
      AUFM~MATNR
      AUFM~WERKS
      AUFM~CHARG
      AUFM~BWART
      AUFM~AUFNR
      AUFM~MENGE
      AUFM~MEINS
      MSEG~SMBLN
      MSEG~SJAHR
      MSEG~SMBLP
      APPENDING CORRESPONDING FIELDS OF TABLE IT_BATCH_JC
      FROM AUFM
      JOIN MSEG ON AUFM~MANDT  =  MSEG~MANDT  AND
                   AUFM~MBLNR  =  MSEG~MBLNR  AND
                   AUFM~MJAHR  =  MSEG~MJAHR
      FOR ALL ENTRIES IN IT_BATCH_JR
      WHERE MSEG~SMBLN = IT_BATCH_JR-MBLNR
        AND MSEG~SJAHR = IT_BATCH_JR-MJAHR %_HINTS ORACLE 'INDEX("MESG" "MSEG~Z10")'
    " Eliminate batch cancelation
    LOOP AT IT_BATCH_JC.
      LOOP AT IT_BATCH_JR WHERE MBLNR = IT_BATCH_JC-SMBLN AND MJAHR = IT_BATCH_JC-SJ
        DELETE IT_BATCH_JR.
      ENDLOOP.
    ENDLOOP.
    IF IT_BATCH_JR[] IS INITIAL.
      EXIT.
    ENDIF.
    READ TABLE IT_BATCH_JR INDEX 1.
    IF SY-SUBRC EQ 0.
      MATNR_J = IT_BATCH_JR-MATNR.
      CHARG_J = IT_BATCH_JR-CHARG.
    ENDIF.
    SELECT SINGLE ABLAD INTO V_ABLAD FROM AFPO WHERE AUFNR EQ IT_AFPO-AUFNR AND ABLA
    IF V_ABLAD EQ 'SS'.
      CALL FUNCTION 'ZQM_GET_BATCH_JR_BY_SR'
        EXPORTING
          CHARG_S = CHARG_J
        IMPORTING
          MATNR_J = MATNR_J
          CHARG_J = CHARG_J.
    ENDIF.
  ENDLOOP.
ENDFUNCTION.
